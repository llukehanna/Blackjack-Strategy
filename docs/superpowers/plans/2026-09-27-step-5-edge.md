# Step 5 — Edge Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Replace the additive house-edge model with Wizard of Odds' own calculator table, and build the Edge module: a live house-edge calculator with a rating, a per-rule breakdown and "Use these rules for training".

**Architecture:** A Node script extracts WoO's "basic strategy with cut card" house edge for all 5,760 supported rule combinations into generated Swift (`WoOEdgeData`). `EdgeCalculator` looks values up in that table, derives 2:1 payouts exactly, adds early surrender from WoO's rule-variation figures, and attributes the total to rules step by step from WoO's baseline. The app gets `Features/Edge`: a thin `EdgeViewModel`, `EdgeText` for all copy, and `EdgeView`, which has a pinned result bar, the breakdown, the shared `RulesForm` and an apply button. It also gets one new Felt component, `EdgeContributionRow`.

**Tech Stack:** Swift 6.2, SwiftUI, iOS 18+, Swift Testing, XCTest (UI), XcodeGen, Node 24 (extractor only).

**Spec:** `docs/superpowers/specs/2026-09-27-step-5-edge-design.md`. Parent spec: `docs/superpowers/specs/2026-09-23-bjs-rebuild-design.md` (§4 design system, §5 Edge, §7 testing). Read both before starting.

Create `tools/woo-edge/extract.js`:

```javascript
#!/usr/bin/env node
// Regenerates the Wizard of Odds house-edge data used by BJSCore's EdgeCalculator.
//
// Downloads WoO's blackjack house edge calculator page, evaluates its inline data script
// (edgeTable + CalculateResult) against a stub form, calls WoO's own CalculateResult() for
// every rule combination the app supports, and reads the "Basic strategy with cut card" box
// (RESULTS7). Writes BJSCore/Sources/BJSCore/Edge/WoOEdgeData.swift.
//
// Usage (from the repo root): node tools/woo-edge/extract.js [repoRoot]
// Dev-only: this executes the page's own script. Review the diff before committing.

const fs = require('fs');
const path = require('path');

const URL = 'https://wizardofodds.com/games/blackjack/calculator/';
const root = process.argv[2] || process.cwd();
const out = path.join(root, 'BJSCore/Sources/BJSCore/Edge/WoOEdgeData.swift');

// The app's deck counts and their radio index on WoO's page (WoO also offers 5 decks, index 3).
const DECKS = [[1, 0], [2, 1], [4, 2], [6, 4], [8, 5]];

function radios(n) {
  return Array.from({ length: n }, () => ({ checked: false }));
}

async function main() {
  const res = await fetch(URL, { headers: { 'User-Agent': 'Mozilla/5.0' } });
  if (!res.ok) throw new Error(`HTTP ${res.status}`);
  const html = await res.text();
  const marker = html.indexOf('var edgeTable');
  if (marker < 0) throw new Error('edgeTable not found; page layout changed');
  const start = html.lastIndexOf('<script', marker);
  const src = html.slice(html.indexOf('>', start) + 1, html.indexOf('</script>', marker));
  if (!src.includes('function CalculateResult')) throw new Error('CalculateResult not found; page layout changed');

  const form = {
    SELECT1: radios(6), SELECT2: radios(2), SELECT3: radios(2), SELECT4: radios(3), SELECT5: radios(3),
    SELECT6: radios(2), SELECT7: radios(2), SELECT8: radios(2), SELECT9: radios(2), SELECT10: radios(2),
    RESULTS6: {}, RESULTS7: {}, RESULTS8: {},
  };
  const document = { selectForm: form };
  const window = {};
  const calculate = new Function('document', 'window', `${src}\nreturn CalculateResult;`)(document, window);

  function select(name, index) {
    form[name].forEach((r, i) => { r.checked = i === index; });
  }

  // Index order, outermost first (must match EdgeCalculator.tableIndex):
  // decks(1,2,4,6,8) · soft17(stands,hits) · das(no,yes) · double(any two,9-11,10-11) ·
  // hands(2,3,4) · rsa(no,yes) · hsa(no,yes) · hole card(peek,none) · surrender(none,late) ·
  // payout(3:2,6:5)
  const values = [];
  for (const [, deckIndex] of DECKS)
    for (let h17 = 0; h17 < 2; h17++)
      for (let das = 0; das < 2; das++)
        for (let dbl = 0; dbl < 3; dbl++)
          for (let hands = 0; hands < 3; hands++)
            for (let rsa = 0; rsa < 2; rsa++)
              for (let hsa = 0; hsa < 2; hsa++)
                for (let enhc = 0; enhc < 2; enhc++)
                  for (let ls = 0; ls < 2; ls++)
                    for (let p65 = 0; p65 < 2; p65++) {
                      select('SELECT1', deckIndex);
                      select('SELECT2', h17);
                      select('SELECT3', das);
                      select('SELECT4', dbl);
                      select('SELECT5', hands);
                      select('SELECT6', rsa);
                      select('SELECT7', hsa);
                      select('SELECT8', 1 - enhc); // WoO: "loses only original bet vs BJ" No = no hole card
                      select('SELECT9', ls);
                      select('SELECT10', p65);
                      calculate();
                      const text = String(form.RESULTS7.value);
                      const value = parseFloat(text);
                      if (!text.endsWith('%') || !Number.isFinite(value)) throw new Error(`bad value ${text}`);
                      values.push(value);
                    }
  if (values.length !== 5760) throw new Error(`expected 5760 values, got ${values.length}`);

  const date = new Date().toLocaleDateString('en-CA');
  const lines = [];
  for (let i = 0; i < values.length; i += 8) {
    lines.push('        ' + values.slice(i, i + 8).map(v => String(v)).join(', ') + ',');
  }
  const swift = `// Generated by tools/woo-edge/extract.js. Do not edit by hand.
// Source: ${URL} (retrieved ${date}).
// House edge by WizardOfOdds.com.

/// Wizard of Odds house edge (%) for total-dependent basic strategy dealt with a cut card,
/// one value per rule combination. Index order, outermost first: decks (1, 2, 4, 6, 8) ·
/// soft 17 (stands, hits) · DAS (no, yes) · double (any two, 9-11, 10-11) · hands (2, 3, 4) ·
/// RSA (no, yes) · hit split aces (no, yes) · hole card (peek, none) · surrender (none, late) ·
/// payout (3:2, 6:5). See \`EdgeCalculator.tableIndex\`.
enum WoOEdgeData {
    static let cutCard: [Double] = [
${lines.join('\n')}
    ]
}
`;
  fs.writeFileSync(out, swift);
  console.log(`wrote ${values.length} values to ${out}`);
}

main().catch(e => { console.error(e); process.exit(1); });
```

/// A rule that moves the house edge away from WoO's baseline game. Cases are listed in
/// attribution order (see `EdgeCalculator.analyze(rules:)`).
public enum EdgeFactor: Sendable, Hashable {
    case decks(BlackjackRules.DeckCount)
    case dealerHitsSoft17
    case payout(BlackjackRules.BlackjackPayout)
    case noDoubleAfterSplit
    case doubleRestriction(BlackjackRules.DoubleRestriction)
    case maxSplitHands(Int)
    case resplitAces
    case hitSplitAces
    case noHoleCard
    /// `pricedAsEarly` is true for late surrender under no hole card, which `RoundEngine`
    /// settles like early surrender (it honours the surrender even against a natural).
    case surrender(BlackjackRules.SurrenderRule, pricedAsEarly: Bool)
}

/// One step of the breakdown: how much this rule changes the house edge, in percentage points
/// (positive is worse for the player), given every rule before it in attribution order.
public struct RuleContribution: Sendable, Hashable {
    public let factor: EdgeFactor
    public let edgeChange: Double

    public init(factor: EdgeFactor, edgeChange: Double) {
        self.factor = factor
        self.edgeChange = edgeChange
    }
}

/// The house edge for a rule set, with its breakdown from WoO's baseline game.
public struct EdgeResult: Sendable, Hashable {
    /// House edge in percent (0.43 means 0.43%). Negative means the player has the edge.
    public let houseEdge: Double
    /// Steps from `EdgeCalculator.baselineRules`. They sum exactly to
    /// `houseEdge - EdgeCalculator.baselineHouseEdge`.
    public let contributions: [RuleContribution]

    public init(houseEdge: Double, contributions: [RuleContribution]) {
        self.houseEdge = houseEdge
        self.contributions = contributions
    }
}

/// House edge for total-dependent basic strategy dealt with a cut card, from Wizard of Odds'
/// house edge calculator (`WoOEdgeData`, generated by `tools/woo-edge/extract.js`).
///
/// Two rules the calculator doesn't offer are derived:
/// - 2:1 payouts, exactly: the payout is linear in unpushed player blackjacks and basic
///   strategy ignores it, so 2:1 (+0.5 units per blackjack against 3:2) sits 5/3 of the
///   3:2 → 6:5 gap (−0.3 units) on the other side of 3:2.
/// - Early surrender: WoO's rule-variation figures (8 decks), not deck-specific.
public struct EdgeCalculator: Sendable {

    /// WoO rule variations (8 decks): early surrender against an ace +0.39% and against a ten
    /// +0.24% for the player.
    public static let earlySurrenderGain = 0.63

    /// WoO's baseline game: 8 decks, S17, DAS, double any two, split to 4 hands, no RSA,
    /// no hit split aces, peek, no surrender, 3:2.
    public static let baselineRules: BlackjackRules = {
        var rules = BlackjackRules()
        rules.deckCount = .eight
        return rules
    }()

    public static let baselineHouseEdge: Double = EdgeCalculator().houseEdge(for: baselineRules)

    public init() {}

    public func houseEdge(for rules: BlackjackRules) -> Double {
        let surrender = Self.effectiveSurrender(rules)
        func table(sixToFive: Bool) -> Double {
            WoOEdgeData.cutCard[Self.tableIndex(rules, lateSurrender: surrender == .late, sixToFive: sixToFive)]
        }
        var edge: Double
        switch rules.blackjackPayout {
        case .threeToTwo:
            edge = table(sixToFive: false)
        case .sixToFive:
            edge = table(sixToFive: true)
        case .twoToOne:
            let threeToTwo = table(sixToFive: false)
            edge = threeToTwo - 5.0 / 3.0 * (table(sixToFive: true) - threeToTwo)
        }
        if surrender == .early { edge -= Self.earlySurrenderGain }
        return edge
    }

    /// The house edge plus its breakdown. Starting from `baselineRules`, the user's rules are
    /// applied one field at a time in `EdgeFactor` order; each changed field records the change
    /// in house edge at that step.
    public func analyze(rules: BlackjackRules) -> EdgeResult {
        let base = Self.baselineRules
        var current = base
        var edge = houseEdge(for: current)
        var contributions: [RuleContribution] = []

        func step(_ changed: Bool, _ factor: EdgeFactor, _ apply: (inout BlackjackRules) -> Void) {
            guard changed else { return }
            apply(&current)
            let next = houseEdge(for: current)
            contributions.append(RuleContribution(factor: factor, edgeChange: next - edge))
            edge = next
        }

        step(rules.deckCount != base.deckCount, .decks(rules.deckCount)) { $0.deckCount = rules.deckCount }
        step(rules.dealerSoft17 != base.dealerSoft17, .dealerHitsSoft17) { $0.dealerSoft17 = rules.dealerSoft17 }
        step(rules.blackjackPayout != base.blackjackPayout, .payout(rules.blackjackPayout)) {
            $0.blackjackPayout = rules.blackjackPayout
        }
        step(rules.doubleAfterSplit != base.doubleAfterSplit, .noDoubleAfterSplit) {
            $0.doubleAfterSplit = rules.doubleAfterSplit
        }
        step(rules.doubleRestriction != base.doubleRestriction, .doubleRestriction(rules.doubleRestriction)) {
            $0.doubleRestriction = rules.doubleRestriction
        }
        step(rules.maxSplitHands != base.maxSplitHands, .maxSplitHands(rules.maxSplitHands)) {
            $0.maxSplitHands = rules.maxSplitHands
        }
        step(rules.resplitAces != base.resplitAces, .resplitAces) { $0.resplitAces = rules.resplitAces }
        step(rules.hitSplitAces != base.hitSplitAces, .hitSplitAces) { $0.hitSplitAces = rules.hitSplitAces }
        step(rules.peekRule != base.peekRule, .noHoleCard) { $0.peekRule = rules.peekRule }
        let lateAsEarly = rules.surrenderRule == .late && rules.peekRule == .europeanNoPeek
        step(rules.surrenderRule != base.surrenderRule, .surrender(rules.surrenderRule, pricedAsEarly: lateAsEarly)) {
            $0.surrenderRule = rules.surrenderRule
        }

        return EdgeResult(houseEdge: edge, contributions: contributions)
    }

    // MARK: - Table

    /// Late surrender under no hole card settles like early surrender in `RoundEngine`.
    static func effectiveSurrender(_ rules: BlackjackRules) -> BlackjackRules.SurrenderRule {
        rules.surrenderRule == .late && rules.peekRule == .europeanNoPeek ? .early : rules.surrenderRule
    }

    /// Index into `WoOEdgeData.cutCard`. The order must match `tools/woo-edge/extract.js`:
    /// decks · soft 17 · DAS · double · hands · RSA · HSA · hole card · late surrender · 6:5.
    static func tableIndex(_ rules: BlackjackRules, lateSurrender: Bool, sixToFive: Bool) -> Int {
        let decks: Int = switch rules.deckCount {
        case .one: 0
        case .two: 1
        case .four: 2
        case .six: 3
        case .eight: 4
        }
        let double: Int = switch rules.doubleRestriction {
        case .anyTwo: 0
        case .nineToEleven: 1
        case .tenToEleven: 2
        }
        let hands = min(max(rules.maxSplitHands, 2), 4) - 2
        var index = decks
        index = index * 2 + (rules.dealerSoft17 == .hits ? 1 : 0)
        index = index * 2 + (rules.doubleAfterSplit ? 1 : 0)
        index = index * 3 + double
        index = index * 3 + hands
        index = index * 2 + (rules.resplitAces ? 1 : 0)
        index = index * 2 + (rules.hitSplitAces ? 1 : 0)
        index = index * 2 + (rules.peekRule == .europeanNoPeek ? 1 : 0)
        index = index * 2 + (lateSurrender ? 1 : 0)
        index = index * 2 + (sixToFive ? 1 : 0)
        return index
    }
}

        + "Early surrender uses an 8-deck estimate."

    static func headline(isPlayerEdge: Bool) -> String {
        isPlayerEdge ? "Player edge" : "House edge"
    }

    /// The pinned number: the absolute edge, "0.64%". The headline label carries the sign.
    static func headlineNumber(_ edge: Double) -> String {
        twoDecimals(abs(hundredths(edge))) + "%"
    }

    /// A signed edge for the breakdown: "0.45%", "−0.03%".
    static func edge(_ edge: Double) -> String {
        let h = hundredths(edge)
        return (h < 0 ? minus : "") + twoDecimals(abs(h)) + "%"
    }

    /// A change in house edge: "+0.22%", "−0.19%", "0.00%".
    static func signedChange(_ change: Double) -> String {
        let h = hundredths(change)
        return (h > 0 ? "+" : h < 0 ? minus : "") + twoDecimals(abs(h)) + "%"
    }

    static func comparison(_ delta: Double) -> String {
        "\(signedChange(delta)) vs your training rules"
    }

    static func label(for factor: EdgeFactor) -> String {
        switch factor {
        case .decks(let count): return count.label
        case .dealerHitsSoft17: return "Dealer hits soft 17"
        case .payout(let payout): return "Blackjack pays \(payout.label)"
        case .noDoubleAfterSplit: return "No double after split"
        case .doubleRestriction(let restriction): return "Double on \(restriction.label)"
        case .maxSplitHands(let hands): return "Split to \(hands) hands"
        case .resplitAces: return "Resplit aces"
        case .hitSplitAces: return "Hit split aces"
        case .noHoleCard: return "No hole card"
        case .surrender(let rule, let pricedAsEarly):
            switch rule {
            case .none: return "No surrender"
            case .late: return pricedAsEarly ? "Late surrender (settles as early)" : "Late surrender"
            case .early: return "Early surrender"
            }
        }
    }

    static func accessibility(for factor: EdgeFactor, change: Double) -> String {
        let h = hundredths(change)
        let effect = h == 0
            ? "doesn't change the house edge"
            : "\(h > 0 ? "raises" : "lowers") the house edge by \(twoDecimals(abs(h))) percent"
        return "\(label(for: factor)), \(effect)"
    }

    static func confirmMessage(summary: String) -> String {
        "New drills will use \(summary). Saved sessions keep their own rules."
    }

    private static func hundredths(_ value: Double) -> Int {
        Int((value * 100).rounded())
    }

    /// 64 → "0.64", 170 → "1.70", 5 → "0.05". Expects a non-negative value.
    private static func twoDecimals(_ hundredths: Int) -> String {
        let fraction = hundredths % 100
        return "\(hundredths / 100).\(fraction < 10 ? "0" : "")\(fraction)"
    }
}

import SwiftUI

enum EdgeContributionLayout {
    /// The bar's share of its half-width: |change| / scale, clamped to 0...1.
    static func barFraction(change: Double, scale: Double) -> Double {
        guard scale > 0 else { return 0 }
        return min(abs(change) / scale, 1)
    }
}

/// One step of a house-edge breakdown: the rule's label and signed change, above a bar that
/// diverges from a centre zero line. It grows right in `incorrect` when the rule raises the house
/// edge (worse for the player), and left in `correct` when it lowers it. `scale` is the largest
/// |change| on screen, so bars compare within one breakdown. Sits inside a `SettingsSection`.
struct EdgeContributionRow: View {
    let label: String
    let value: String
    let change: Double
    let scale: Double
    let accessibilityText: String

    var body: some View {
        VStack(alignment: .leading, spacing: FeltSpacing.xs) {
            HStack(alignment: .firstTextBaseline, spacing: FeltSpacing.m) {
                Text(label)
                    .feltText(.body)
                    .foregroundStyle(FeltColor.textPrimary)
                    .fixedSize(horizontal: false, vertical: true)
                Spacer(minLength: FeltSpacing.s)
                Text(value)
                    .feltText(.body)
                    .monospacedDigit()
                    .foregroundStyle(FeltColor.textSecondary)
            }
            EdgeContributionBar(fraction: EdgeContributionLayout.barFraction(change: change, scale: scale),
                                raisesEdge: change > 0)
                .frame(height: 6)
        }
        .padding(.horizontal, FeltSpacing.l)
        .padding(.vertical, FeltSpacing.s)
        .frame(minHeight: FeltTapTarget.minimum)
        .accessibilityElement(children: .ignore)
        .accessibilityLabel(accessibilityText)
    }
}

private struct EdgeContributionBar: View {
    let fraction: Double
    let raisesEdge: Bool

    var body: some View {
        GeometryReader { geo in
            let half = geo.size.width / 2
            let length = half * fraction
            ZStack(alignment: .leading) {
                Capsule()
                    .fill(FeltColor.textTertiary.opacity(0.2))
                    .frame(height: 2)
                    .frame(maxHeight: .infinity)
                Rectangle()
                    .fill(FeltColor.textTertiary.opacity(0.6))
                    .frame(width: 1)
                    .offset(x: half - 0.5)
                Capsule()
                    .fill(raisesEdge ? FeltColor.incorrect : FeltColor.correct)
                    .frame(width: length)
                    .offset(x: raisesEdge ? half : half - length)
            }
        }
    }
}

import SwiftUI
import BJSCore

/// The Edge module: the house edge for any rules, live. It opens on the active rules; edits are
/// discarded on close unless applied. The result bar stays pinned above the scrolling content.
struct EdgeView: View {
    let onClose: () -> Void
    @Environment(ActiveRulesStore.self) private var rulesStore

    var body: some View {
        // The model is created once from the rules active at open (see EdgeScreen's @State).
        EdgeScreen(activeRules: rulesStore.rules, onClose: onClose)
    }
}

private struct EdgeScreen: View {
    let onClose: () -> Void
    @Environment(ActiveRulesStore.self) private var rulesStore
    @State private var model: EdgeViewModel
    @State private var confirmingApply = false
    @State private var showsToast = false

    init(activeRules: BlackjackRules, onClose: @escaping () -> Void) {
        self.onClose = onClose
        _model = State(initialValue: EdgeViewModel(activeRules: activeRules))
    }

    var body: some View {
        @Bindable var model = model
        let result = model.result
        ZStack {
            FeltBackground()
            VStack(spacing: 0) {
                resultBar(result)
                    .padding(.horizontal, FeltSpacing.l)
                    .padding(.bottom, FeltSpacing.m)
                Rectangle()
                    .fill(FeltColor.textTertiary.opacity(0.25))
                    .frame(height: 0.5)
                ScrollView {
                    VStack(alignment: .leading, spacing: FeltSpacing.xl) {
                        breakdown(result)
                        RulesForm(rules: $model.rules)
                        applySection
                        Text(EdgeText.footnote)
                            .feltText(.body)
                            .foregroundStyle(FeltColor.textTertiary)
                            .fixedSize(horizontal: false, vertical: true)
                    }
                    .frame(maxWidth: .infinity, alignment: .leading)
                    .padding(FeltSpacing.l)
                }
            }
        }
        .overlay(alignment: .bottom) {
            if showsToast {
                FeltToast(text: EdgeText.toast)
                    .fixedSize()
                    .padding(.bottom, FeltSpacing.xl)
                    .transition(.opacity)
            }
        }
        .confirmationDialog(EdgeText.confirmTitle, isPresented: $confirmingApply, titleVisibility: .visible) {
            Button(EdgeText.confirmButton) {
                model.apply(to: rulesStore)
                flashToast()
            }
        } message: {
            Text(EdgeText.confirmMessage(summary: RulesSummary.text(for: model.rules)))
        }
    }

    private func resultBar(_ result: EdgeResult) -> some View {
        VStack(alignment: .leading, spacing: FeltSpacing.xs) {
            HStack {
                CloseButton(identifier: "edge.close", action: onClose)
                Spacer()
                Text(EdgeText.headline(isPlayerEdge: model.isPlayerEdge))
                    .feltText(.label)
                    .foregroundStyle(FeltColor.textTertiary)
            }
            HStack(alignment: .firstTextBaseline, spacing: FeltSpacing.m) {
                Text(EdgeText.headlineNumber(result.houseEdge))
                    .feltText(.display)
                    .monospacedDigit()
                    .foregroundStyle(FeltColor.textPrimary)
                    .accessibilityIdentifier("edge.number")
                Text(model.rating.displayName)
                    .feltText(.body)
                    .foregroundStyle(ratingColor(model.rating))
                    .padding(.horizontal, FeltSpacing.m)
                    .padding(.vertical, FeltSpacing.xs)
                    .background(FeltColor.surfaceInset, in: Capsule())
                    .accessibilityIdentifier("edge.rating")
                Spacer(minLength: 0)
            }
            if let comparison = model.comparison {
                Text(EdgeText.comparison(comparison))
                    .feltText(.body)
                    .foregroundStyle(FeltColor.textSecondary)
                    .accessibilityIdentifier("edge.comparison")
            }
        }
    }

    private func breakdown(_ result: EdgeResult) -> some View {
        let scale = model.breakdownScale
        return SettingsSection(title: EdgeText.breakdownTitle) {
            SettingsRow(label: EdgeText.baselineLabel) {
                Text(EdgeText.edge(EdgeCalculator.baselineHouseEdge)).feltText(.body).monospacedDigit()
            }
            ForEach(result.contributions, id: \.factor) { contribution in
                EdgeContributionRow(
                    label: EdgeText.label(for: contribution.factor),
                    value: EdgeText.signedChange(contribution.edgeChange),
                    change: contribution.edgeChange,
                    scale: scale,
                    accessibilityText: EdgeText.accessibility(for: contribution.factor,
                                                              change: contribution.edgeChange))
            }
            SettingsRow(label: EdgeText.totalLabel) {
                Text(EdgeText.edge(result.houseEdge)).feltText(.body).monospacedDigit()
            }
        }
    }

    @ViewBuilder
    private var applySection: some View {
        if model.canApply {
            PrimaryButton(title: EdgeText.applyTitle) { confirmingApply = true }
                .accessibilityIdentifier("edge.apply")
        } else {
            Text(EdgeText.appliedCaption)
                .feltText(.body)
                .foregroundStyle(FeltColor.textSecondary)
                .frame(maxWidth: .infinity, minHeight: 52)
                .accessibilityIdentifier("edge.applied")
        }
    }

    private func ratingColor(_ rating: EdgeRating) -> Color {
        switch rating {
        case .good: return FeltColor.correct
        case .ok: return FeltColor.textSecondary
        case .poor: return FeltColor.incorrect
        }
    }

    private func flashToast() {
        withAnimation(FeltMotion.ui) { showsToast = true }
        AccessibilityNotification.Announcement(EdgeText.toast).post()
        Task {
            try? await Task.sleep(for: .seconds(FeltToast.displayDuration))
            withAnimation(FeltMotion.ui) { showsToast = false }
        }
    }
}
