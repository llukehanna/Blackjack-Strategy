```swift
// BJS/Domain/WhyExplanation.swift

public struct WhyContext: Identifiable, Equatable, Sendable {
    public let id = UUID()
    public let handTotal: Int
    public let handType: HandType            // .hard, .soft, .pair
    public let dealerUpCard: Rank
    public let userAction: PlayerAction
    public let correctAction: PlayerAction
    public let rules: CasinoRules
}

public enum WhyExplanation {
    /// Returns a short (1–3 sentence) explanation of why `correctAction` is
    /// the right move for the given context. Pure function; no side effects.
    public static func explain(_ context: WhyContext) -> String
}
```

  <action>
1. Create `BJS/Domain/WhyExplanation.swift`:
   - Define `WhyContext` struct exactly as in the <interfaces> block above. Use the existing BJSCore types (`HandType`, `Rank`, `PlayerAction`, `CasinoRules`) — import BJSCore. If a type doesn't exist with that exact name, use the closest equivalent already in BJSCore and adjust the struct to match.
   - Define `enum WhyExplanation` with a single static method `explain(_ context: WhyContext) -> String`.
   - Implementation strategy: switch on `(context.handType, context.correctAction)` to pick a template family, then interpolate `context.handTotal`, `context.dealerUpCard`, and any rule modifiers. Examples:
     - `(.hard, .stand)` where total >= 12 and dealer up is 2–6 → "Hard \(total) vs dealer \(up): the dealer is likely to bust with a weak up card, so standing lets the dealer beat themselves."
     - `(.hard, .stand)` where total >= 17 → "Hard \(total): you're already in a strong stand range — hitting risks busting for almost no upside."
     - `(.hard, .hit)` where total <= 11 → "Hard \(total): you can't bust, so always take a free card."
     - `(.hard, .hit)` where 12...16 vs dealer 7+ → "Hard \(total) vs dealer \(up): the dealer's strong up card will likely beat you if you stand. Hitting is the lesser-of-two-evils play."
     - `(.hard, .double)` for 9, 10, 11 vs weak dealer → "Hard \(total) vs dealer \(up): doubling capitalizes on the dealer's weak up card with a single high-value card."
     - `(.soft, .hit)` → "Soft \(total): you can't bust on the next card, so hitting only improves your hand."
     - `(.soft, .stand)` for soft 19/20 → "Soft \(total): already a strong total — locking it in beats risking a worse one."
     - `(.soft, .double)` → "Soft \(total) vs dealer \(up): double for value — you can't bust and the dealer is weak."
     - `(.pair, .split)` for 8s/Aces → "Pair of \(rank)s: \(total) is a losing hand; splitting turns it into two fresh hands starting with a strong card."
     - `(.pair, .stand)` for 10s → "Pair of 10s: 20 is one of the strongest totals in the game — never split a winning hand."
   - Provide a sensible default branch covering any (handType, action) the switch missed so the function never returns "".
