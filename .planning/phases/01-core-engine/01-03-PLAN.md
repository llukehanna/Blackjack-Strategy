       ```swift
       struct HiLoCounter: Sendable {
           private(set) var runningCount: Int = 0

           mutating func process(_ card: Card) {
               runningCount += card.rank.hiLoValue
           }

           mutating func process(_ cards: [Card]) {
               for card in cards {
                   runningCount += card.rank.hiLoValue
               }
           }

           func trueCount(decksRemaining: Double) -> Double {
               guard decksRemaining > 0 else { return 0 }
               return Double(runningCount) / decksRemaining
           }

           mutating func reset() {
               runningCount = 0
           }
       }
       ```

       This is intentionally simple. The `trueCount` method takes `decksRemaining` as a parameter (from `Shoe.decksRemaining`) rather than holding a reference to a Shoe, keeping the counter a pure value type.

    1. Create `Sources/BJSCore/Edge/EdgeCalculator.swift`:
       ```swift
       struct EdgeCalculator: Sendable {
           /// Baseline: 8D S17 DAS, no surrender, no RSA, no hit-split-aces,
           /// any-two doubles, American peek, 3:2 BJ, max 4 split hands.
           /// House edge: 0.43%
           static let baselineHouseEdge: Double = 0.43

           struct RuleContribution: Sendable {
               let rule: String
               let delta: Double  // Negative = worse for player, positive = better for player
           }

           struct EdgeResult: Sendable {
               let houseEdge: Double  // As percentage (e.g., 0.43 means 0.43%)
               let contributions: [RuleContribution]
           }

           func analyze(rules: BlackjackRules) -> EdgeResult {
               var contributions: [RuleContribution] = []

               // Deck count delta (baseline is 8 decks)
               let deckDelta = Self.deckCountDelta(rules.deckCount)
               if abs(deckDelta) > 0.001 {
                   contributions.append(RuleContribution(
                       rule: "\(rules.deckCount.rawValue) deck(s) vs 8-deck baseline",
                       delta: deckDelta
                   ))
               }

               // Dealer soft 17 (baseline is S17)
               if rules.dealerSoft17 == .hits {
                   contributions.append(RuleContribution(rule: "Dealer hits soft 17", delta: -0.22))
               }

               // DAS (baseline has DAS)
               if !rules.doubleAfterSplit {
                   contributions.append(RuleContribution(rule: "No double after split", delta: -0.14))
               }

               // RSA (baseline has no RSA)
               if rules.resplitAces {
                   contributions.append(RuleContribution(rule: "Resplit aces allowed", delta: 0.08))
               }

               // Hit split aces (baseline has no hit-split-aces)
               if rules.hitSplitAces {
                   contributions.append(RuleContribution(rule: "Hit split aces allowed", delta: 0.19))
               }

               // Surrender (baseline has no surrender)
               switch rules.surrenderRule {
               case .none: break
               case .late:
                   contributions.append(RuleContribution(rule: "Late surrender", delta: 0.08))
               case .early:
                   contributions.append(RuleContribution(rule: "Early surrender", delta: 0.24))
               }

               // Double restriction (baseline is any two)
               switch rules.doubleRestriction {
               case .anyTwo: break
               case .nineToEleven:
                   contributions.append(RuleContribution(rule: "Double on 9-11 only", delta: -0.09))
               case .tenToEleven:
                   contributions.append(RuleContribution(rule: "Double on 10-11 only", delta: -0.18))
               }

               // Peek rule (baseline is American peek)
               if rules.peekRule == .europeanNoPeek {
                   contributions.append(RuleContribution(rule: "European no hole card", delta: -0.11))
               }

               // BJ payout (baseline is 3:2)
               switch rules.blackjackPayout {
               case .threeToTwo: break
               case .sixToFive:
                   contributions.append(RuleContribution(rule: "Blackjack pays 6:5", delta: -1.39))
               case .twoToOne:
                   contributions.append(RuleContribution(rule: "Blackjack pays 2:1", delta: 0.32))
               case .evenMoney:
                   contributions.append(RuleContribution(rule: "Blackjack pays even money", delta: -2.27))
               }

               // Max split hands (baseline is 4)
               switch rules.maxSplitHands {
               case 2:
                   contributions.append(RuleContribution(rule: "Split to 2 hands max", delta: -0.10))
               case 3:
                   contributions.append(RuleContribution(rule: "Split to 3 hands max", delta: -0.01))
               default: break  // 4 is baseline
               }

               let totalDelta = contributions.reduce(0.0) { $0 + $1.delta }
               // House edge: baseline minus player-favorable deltas
               // Deltas are expressed as player return effect:
               //   positive delta = better for player = lower house edge
               //   negative delta = worse for player = higher house edge
               let houseEdge = Self.baselineHouseEdge - totalDelta

               return EdgeResult(houseEdge: houseEdge, contributions: contributions)
           }

           func houseEdge(for rules: BlackjackRules) -> Double {
               analyze(rules: rules).houseEdge
           }

           private static func deckCountDelta(_ deckCount: BlackjackRules.DeckCount) -> Double {
               switch deckCount {
               case .one: return 0.48
               case .two: return 0.19
               case .four: return 0.06
               case .six: return 0.02
               case .eight: return 0.0  // baseline
               }
           }
       }
       ```

       Note on delta signs: Per WoO, deltas are expressed as "Effect on Player Return." A positive value means the rule favors the player (lowers house edge). The house edge = baseline - sum(deltas). Example: 6 decks has +0.02 player return delta vs 8 decks, so house edge = 0.43 - 0.02 = 0.41. But the expected value for 6D S17 DAS is 0.40%, so the baseline + deck delta should produce approximately 0.41% which rounds to 0.40% within tolerance. The exact values should match the expected test values within 0.02% tolerance.

       **Important for 2:1 BJ payout delta:** This is a rare rule where BJ pays more than 3:2. The delta is approximately +0.32% (better for player). Verify this value is reasonable.
